All tests pass. Refactored to a single-pass algorithm:

```javascript
function findLatestAndCount(events, type) {
  let count = 0;
  let latest = null;
  for (const event of events) {
    if (event && event.type === type && Number.isFinite(event.timestamp)) {
      count++;
      if (!latest || event.timestamp > latest.timestamp) {
        latest = event;
      }
    }
  }
  return { count, latest };
}

export function countByType(events, type) {
  return findLatestAndCount(events, type).count;
}

export function latestByType(events, type) {
  return findLatestAndCount(events, type).latest;
}
```
